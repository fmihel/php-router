<?php

class SodiumExchange
{
    private string $appSecret;

    public function __construct(string $appSecret)
    {
        $this->appSecret = $appSecret;
    }

    /**
     * Детерминированная генерация пары ключей на основе JWT-данных
     */
    private function deriveKeypair(string $jwtIdentifier): string
    {
        // Получаем ровно 32 байта (SODIUM_CRYPTO_BOX_SEEDBYTES)
        $seed = hash_hmac('sha256', $jwtIdentifier, $this->appSecret, true);
        
        return sodium_crypto_box_seed_keypair($seed);
    }

    /**
     * Вызывается при авторизации: возвращает публичный ключ для клиента
     */
    public function getPublicKeyForClient(string $jwtIdentifier): string
    {
        $keypair = $this->deriveKeypair($jwtIdentifier);
        $publicKey = sodium_crypto_box_publickey($keypair);
        
        // Очищаем память от приватных данных
        sodium_memzero($keypair);

        return base64_encode($publicKey);
    }

    /**
     * Вызывается при получении зашифрованного запроса от клиента
     */
    public function decryptFromClient(string $encryptedBase64, string $jwtIdentifier): string
    {
        $ciphertext = base64_decode($encryptedBase64, true);
        if ($ciphertext === false) {
            throw new InvalidArgumentException('Invalid base64 payload');
        }

        $keypair = $this->deriveKeypair($jwtIdentifier);
        
        // Расшифровка sealed box
        $decrypted = sodium_crypto_box_seal_open($ciphertext, $keypair);
        
        sodium_memzero($keypair);

        if ($decrypted === false) {
            throw new RuntimeException('Decryption failed or data tampered');
        }

        return $decrypted;
    }
}

import _sodium from 'libsodium-wrappers';

async function encryptPayload(dataObject, serverPublicKeyBase64) {
    // Обязательно дожидаемся инициализации WebAssembly модуля
    await _sodium.ready;
    const sodium = _sodium;

    // 1. Декодируем публичный ключ сервера из Base64 в Uint8Array
    // Используем вариант ORIGINAL, чтобы совпасть с PHP base64_encode()
    const publicKey = sodium.from_base64(
        serverPublicKeyBase64, 
        sodium.base64_variants.ORIGINAL
    );

    // 2. Превращаем данные в строку и затем в байты
    const messageBytes = sodium.from_string(JSON.stringify(dataObject));

    // 3. Шифруем с помощью crypto_box_seal
    const encryptedBytes = sodium.crypto_box_seal(messageBytes, publicKey);

    // 4. Возвращаем в Base64 для отправки по HTTP (JSON / FormData)
    return sodium.to_base64(
        encryptedBytes, 
        sodium.base64_variants.ORIGINAL
    );
}
